Add this at the end of schema.prisma after FormShare model:

model Issue {
  id          String   @id @default(cuid())
  tenantId    String
  projectId   String
  title       String
  description String?
  status      String   @default("backlog") // backlog | todo | in-progress | review | done | closed
  priority    String   @default("medium")  // low | medium | high | critical
  type        String   @default("task")    // bug | feature | task | improvement
  severity    String?  // minor | major | critical | blocker (for bugs)
  assigneeId String?
  reporterId String
  tags        String[] @default([])
  dueDate     DateTime?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  tenant   Tenant  @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  project  Project @relation(fields: [projectId], references: [id], onDelete: Cascade)
  assignee User?   @relation("IssueAssignee", fields: [assigneeId], references: [id], onDelete: SetNull)
  reporter User    @relation("IssueReporter", fields: [reporterId], references: [id], onDelete: Cascade)
  comments Comment[]
  attachments Attachment[]

  @@index([tenantId])
  @@index([projectId])
  @@index([assigneeId])
  @@index([reporterId])
  @@index([status])
  @@index([priority])
  @@map("issues")
}
